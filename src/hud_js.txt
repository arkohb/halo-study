function xpPop(x, y, txt, bad){
  if (reduced()) return;
  var d = el('div','xp', txt);
  if (bad) d.style.color = 'var(--no)';
  d.style.left = (x - 18) + 'px'; d.style.top = (y - 26) + 'px';
  document.body.appendChild(d);
  setTimeout(function(){ if (d.parentNode) d.parentNode.removeChild(d); }, 1200);
}
function feedback(card, ok, btn){
  if (!card) return;
  card.classList.remove('hit','miss');
  if (ok){
    card.classList.add('hit');
    if (btn){ var r = btn.getBoundingClientRect(); xpPop(r.right - 10, r.top, '+10 XP'); }
  } else {
    void card.offsetWidth;
    card.classList.add('miss');
    setTimeout(function(){ card.classList.remove('miss'); }, 600);
  }
}

/* step-by-step reveal of any worked solution */
function fillSol(sol, label, steps, extraHtml){
  sol.innerHTML = '';
  sol.appendChild(el('span','lbl', label));
  var ol = el('ol'), lis = [];
  for (var i = 0; i < steps.length; i++){
    var li = el('li', i === 0 ? '' : 'hid', steps[i]);
    ol.appendChild(li); lis.push(li);
  }
  sol.appendChild(ol);
  var tail = el('div', null, extraHtml || '');
  var many = steps.length > 1;
  if (many && extraHtml) tail.hidden = true;
  var shown = 1;
  if (many){
    var bar = el('div','stepbar');
    var nextB = el('button','ghost','Next step'); nextB.type = 'button';
    var allB  = el('button','ghost','Show all');  allB.type  = 'button';
    var pips = el('div','pips');
    for (var p = 0; p < steps.length; p++) pips.appendChild(el('i', p === 0 ? 'on' : ''));
    function paintPips(){
      var kids = pips.children;
      for (var k = 0; k < kids.length; k++) kids[k].className = (k < shown ? 'on' : '');
      nextB.textContent = shown >= steps.length ? 'All steps shown' : 'Next step (' + shown + '/' + steps.length + ')';
      nextB.disabled = shown >= steps.length;
      if (shown >= steps.length){ tail.hidden = false; allB.hidden = true; }
    }
    function reveal(n){
      while (shown < n && shown < steps.length){
        lis[shown].className = 'pop'; shown++;
      }
      paintPips();
    }
    nextB.addEventListener('click', function(){ reveal(shown + 1); });
    allB.addEventListener('click', function(){ reveal(steps.length); });
    bar.appendChild(nextB); bar.appendChild(allB); bar.appendChild(pips);
    sol.appendChild(bar);
    paintPips();
  }
  sol.appendChild(tail);
}

function gcd(a, b){ a = Math.abs(a); b = Math.abs(b); while (b){ var t = b; b = a % b; a = t; } return a || 1; }

function labNumberLine(){
  var c = labCard('Directed numbers', 'Slide the two numbers and watch the jumps on the line. Adding moves right, subtracting moves left.');
  var a = -4, b = 6, out = el('div','labout'), svg = el('div');
  function draw(){
    var lo = -20, hi = 20, w = 300, pad = 14;
    var sx = function(v){ return pad + (v - lo) / (hi - lo) * (w - 2 * pad); };
    var s = '<svg viewBox="0 0 300 70" class="dia" style="width:100%;max-width:320px">';
    s += '<line x1="' + pad + '" y1="40" x2="' + (w - pad) + '" y2="40" class="dline"/>';
    for (var v = lo; v <= hi; v += 5){
      s += '<line x1="' + sx(v).toFixed(1) + '" y1="35" x2="' + sx(v).toFixed(1) + '" y2="45" class="dline"/>';
      s += '<text x="' + sx(v).toFixed(1) + '" y="60" class="dmut" text-anchor="middle">' + String(v).replace('-', '−') + '</text>';
    }
    var sum = a + b;
    s += '<line x1="' + sx(a).toFixed(1) + '" y1="40" x2="' + sx(sum).toFixed(1) + '" y2="40" class="dacc" stroke-width="4"/>';
    s += '<circle cx="' + sx(a).toFixed(1) + '" cy="40" r="5" class="dopenpt"/>';
    s += '<circle cx="' + sx(sum).toFixed(1) + '" cy="40" r="5" class="dsolidpt"/>';
    s += '<text x="' + sx(sum).toFixed(1) + '" y="22" class="dtxt" text-anchor="middle">' + String(sum).replace('-', '−') + '</text>';
    s += '</svg>';
    svg.innerHTML = s;
    var A = String(a).replace('-', '−'), B = String(b).replace('-', '−');
    out.innerHTML = '(' + A + ') + (' + B + ') = <b>' + String(a + b).replace('-', '−') + '</b><br>'
      + '(' + A + ') − (' + B + ') = <b>' + String(a - b).replace('-', '−') + '</b><br>'
      + '(' + A + ') × (' + B + ') = <b>' + String(a * b).replace('-', '−') + '</b>'
      + '<br><span style="color:var(--ink-2)">' + (b < 0 ? 'Adding a negative moves LEFT.' : 'Adding a positive moves RIGHT.') + '</span>';
  }
  slider(c, 'First number', -15, 15, a, 1, function(v){ a = v; draw(); });
  slider(c, 'Second number', -15, 15, b, 1, function(v){ b = v; draw(); });
  c.appendChild(svg); c.appendChild(out); draw();
  return c;
}

function labFractions(){
  var c = labCard('Fraction bars', 'Build two fractions and see them as bars. The app compares them and adds them for you.');
  var n1 = 2, d1 = 3, n2 = 1, d2 = 4;
  var b1 = el('div','fbar','<i></i><span></span>'), b2 = el('div','fbar','<i></i><span></span>');
  var out = el('div','labout');
  function draw(){
    var v1 = n1 / d1, v2 = n2 / d2;
    b1.querySelector('i').style.width = Math.min(100, v1 * 100) + '%';
    b2.querySelector('i').style.width = Math.min(100, v2 * 100) + '%';
    b1.querySelector('span').textContent = n1 + '/' + d1 + '  =  ' + v1.toFixed(3) + '  =  ' + (v1 * 100).toFixed(1) + '%';
    b2.querySelector('span').textContent = n2 + '/' + d2 + '  =  ' + v2.toFixed(3) + '  =  ' + (v2 * 100).toFixed(1) + '%';
    var sn = n1 * d2 + n2 * d1, sd = d1 * d2, g = gcd(sn, sd);
    var sign = v1 > v2 ? '&gt;' : (v1 < v2 ? '&lt;' : '=');
    out.innerHTML = n1 + '/' + d1 + ' <b>' + sign + '</b> ' + n2 + '/' + d2
      + '<br>sum = ' + sn + '/' + sd + ' = <b>' + (sn / g) + '/' + (sd / g) + '</b>'
      + (sn / g >= sd / g ? ' = ' + Math.floor(sn / sd) + ' ' + (sn % sd ? ((sn % sd) / g) + '/' + (sd / g) : '') : '')
      + '<br><span style="color:var(--ink-2)">L.C.M. of ' + d1 + ' and ' + d2 + ' is ' + (d1 * d2 / gcd(d1, d2)) + '</span>';
  }
  slider(c, 'Numerator 1', 1, 12, n1, 1, function(v){ n1 = v; draw(); });
  slider(c, 'Denominator 1', 1, 12, d1, 1, function(v){ d1 = v; draw(); });
  c.appendChild(b1);
  slider(c, 'Numerator 2', 1, 12, n2, 1, function(v){ n2 = v; draw(); });
  slider(c, 'Denominator 2', 1, 12, d2, 1, function(v){ d2 = v; draw(); });
  c.appendChild(b2); c.appendChild(out); draw();
  return c;
}

function labSpinner(){
  var c = labCard('Probability spinner', 'Set how many beads of each colour, then spin. Watch the experimental result settle towards the theoretical probability.');
  var cnt = [3, 5, 2], names = ['Red','Blue','Green'], cols = ['#E0543C','#3C7FE0','#2FB37A'];
  var tally = [0, 0, 0], wheel = el('div','spinner'), out = el('div','labout'), tal = el('div','tally');
  var angle = 0;
  function paintWheel(){
    var tot = cnt[0] + cnt[1] + cnt[2];
    if (!tot){ wheel.style.background = 'var(--panel)'; return; }
    var a = 0, parts = [];
    for (var i = 0; i < 3; i++){
      var frac = cnt[i] / tot, b = a + frac * 360;
      parts.push(cols[i] + ' ' + a.toFixed(1) + 'deg ' + b.toFixed(1) + 'deg');
      a = b;
    }
    wheel.style.background = 'conic-gradient(' + parts.join(',') + ')';
  }
  function paintOut(pickIdx){
    var tot = cnt[0] + cnt[1] + cnt[2], spins = tally[0] + tally[1] + tally[2];
    var s = '';
    if (pickIdx != null) s += 'Landed on <b>' + names[pickIdx] + '</b><br>';
    for (var i = 0; i < 3; i++){
      if (!cnt[i]) continue;
      var th = tot ? (cnt[i] / tot) : 0;
      var ex = spins ? (tally[i] / spins) : 0;
      s += names[i] + ': P = ' + cnt[i] + '/' + tot + ' = ' + th.toFixed(2)
         + '  |  got ' + tally[i] + '/' + spins + (spins ? ' = ' + ex.toFixed(2) : '') + '<br>';
    }
    out.innerHTML = s || 'Add some beads first.';
    tal.innerHTML = '<span>spins: ' + spins + '</span>';
  }
  for (var i = 0; i < 3; i++){
    (function(k){
      slider(c, names[k] + ' beads', 0, 10, cnt[k], 1, function(v){ cnt[k] = v; paintWheel(); paintOut(null); });
    })(i);
  }
  wheel.style.cursor = 'pointer';
  c.appendChild(el('div','needle'));
  c.appendChild(wheel);
  var go = el('button','btn','Spin'); go.type = 'button';
  go.addEventListener('click', function(){
    var tot = cnt[0] + cnt[1] + cnt[2];
    if (!tot) return;
    var r = Math.random() * tot, acc = 0, pick = 0;
    for (var i = 0; i < 3; i++){ acc += cnt[i]; if (r < acc){ pick = i; break; } }
    tally[pick]++;
    angle += 720 + Math.random() * 360;
    if (!reduced()) wheel.style.transform = 'rotate(' + angle + 'deg)';
    paintOut(pick);
  });
  wheel.addEventListener('click', function(){ go.click(); });
  var row = el('div','labrow'); row.appendChild(go);
  var rst = el('button','ghost','Reset tally'); rst.type = 'button';
  rst.addEventListener('click', function(){ tally = [0,0,0]; paintOut(null); });
  row.appendChild(rst); row.appendChild(tal);
  c.appendChild(row); c.appendChild(out);
  paintWheel(); paintOut(null);
  return c;
}

function labCircuit(){
  var c = labCard('Series or parallel?', 'Tap a lamp to blow its filament, then switch between series and parallel to see what happens.');
  var mode = 'series', blown = [false, false, false];
  var box = el('div','lamps'), out = el('div','labout');
  function draw(){
    box.innerHTML = '';
    var anyBlown = blown[0] || blown[1] || blown[2];
    for (var i = 0; i < 3; i++){
      (function(k){
        var lit = (mode === 'series') ? !anyBlown : !blown[k];
        var l = el('div','lamp' + (lit ? ' lit' : '') + (blown[k] ? ' blown' : ''), lit ? '☀' : '○');
        l.title = blown[k] ? 'Blown — tap to replace' : 'Working — tap to blow';
        l.style.cursor = 'pointer';
        l.addEventListener('click', function(){ blown[k] = !blown[k]; draw(); });
        box.appendChild(l);
      })(i);
    }
    out.innerHTML = '<b>' + (mode === 'series' ? 'Series' : 'Parallel') + '</b> — '
      + (mode === 'series'
          ? (anyBlown ? 'one lamp has blown, so the single path is broken and <b>all the lamps go out</b>.'
                      : 'one path only; the same current passes through every lamp.')
          : (anyBlown ? 'each lamp has its own path, so <b>only the blown lamp goes out</b>.'
                      : 'separate branches; the current divides between them.'))
      + '<br><span style="color:var(--ink-2)">House wiring is parallel, for exactly this reason.</span>';
  }
  var row = el('div','labrow');
  var seg = el('div','seg');
  [['series','Series'],['parallel','Parallel']].forEach(function(p){
    var b = el('button', null, p[1]); b.type = 'button';
    b.setAttribute('aria-pressed', mode === p[0] ? 'true' : 'false');
    b.addEventListener('click', function(){
      mode = p[0];
      var kids = seg.children;
      for (var k = 0; k < kids.length; k++) kids[k].setAttribute('aria-pressed', 'false');
      b.setAttribute('aria-pressed','true');
      draw();
    });
    seg.appendChild(b);
  });
  row.appendChild(seg);
  var fix = el('button','ghost','Replace all lamps'); fix.type = 'button';
  fix.addEventListener('click', function(){ blown = [false,false,false]; draw(); });
  row.appendChild(fix);
  c.appendChild(row); c.appendChild(box); c.appendChild(out); draw();
  return c;
}

function labBinary(){
  var c = labCard('Binary switches', 'Each switch is one bit. Tap them to build a number, or take the challenge.');
  var bits = [0,0,0,0,0,0,0,0];   /* 128 ... 1 */
  var row = el('div','bits'), out = el('div','labout'), target = null;
  function val(){ var v = 0; for (var i = 0; i < 8; i++) v += bits[i] * (1 << (7 - i)); return v; }
  function draw(){
    row.innerHTML = '';
    for (var i = 0; i < 8; i++){
      (function(k){
        var b = el('button','bit' + (bits[k] ? ' on' : ''), bits[k] + '<small>' + (1 << (7 - k)) + '</small>');
        b.type = 'button';
        b.addEventListener('click', function(){ bits[k] = bits[k] ? 0 : 1; draw(); });
        row.appendChild(b);
      })(i);
    }
    var v = val(), s = '';
    for (var j = 0; j < 8; j++) s += bits[j];
    out.innerHTML = 'binary <b>' + s + '</b> = ' + bits.map(function(b, i){ return b ? (1 << (7 - i)) : 0; })
        .filter(function(x){ return x; }).join(' + ') + (v ? ' = ' : '') + '<b>' + v + '</b>'
      + (target != null
          ? '<br>Target: <b>' + target + '</b> — ' + (v === target
              ? '<span style="color:var(--ok)">correct!</span>'
              : (v < target ? 'too small' : 'too big'))
          : '');
    if (target != null && v === target){ target = null; }
  }
  c.appendChild(row);
  var rowb = el('div','labrow');
  var ch = el('button','btn','Challenge me'); ch.type = 'button';
  ch.addEventListener('click', function(){
    target = 1 + Math.floor(Math.random() * 254);
    bits = [0,0,0,0,0,0,0,0]; draw();
  });
  var cl = el('button','ghost','Clear'); cl.type = 'button';
  cl.addEventListener('click', function(){ bits = [0,0,0,0,0,0,0,0]; draw(); });
  rowb.appendChild(ch); rowb.appendChild(cl);
  c.appendChild(rowb); c.appendChild(out); draw();
  return c;
}

function labWords(){
  var SENT = [
    ['Neither', 'the', 'boys', 'nor', 'their', 'teacher', 'was', 'present'],
    ['One', 'of', 'the', 'girls', 'has', 'lost', 'her', 'bag'],
    ['Although', 'it', 'rained', 'heavily', 'we', 'went', 'to', 'school'],
    ['The', 'box', 'of', 'pencils', 'is', 'on', 'the', 'table'],
    ['He', 'asked', 'me', 'whether', 'I', 'had', 'finished', 'the', 'work']
  ];
  var c = labCard('Sentence builder', 'Tap the words in the correct order. These are the concord and structure patterns the examiners test most.');
  var idx = 0, picked = [], pool = [], line = el('div','wline'), box = el('div','words'), out = el('div','labout');
  function shuffle(a){
    a = a.slice();
    for (var i = a.length - 1; i > 0; i--){ var j = Math.floor(Math.random() * (i + 1)), t = a[i]; a[i] = a[j]; a[j] = t; }
    return a;
  }
  function load(){
    picked = []; pool = shuffle(SENT[idx]); draw();
  }
  function draw(){
    line.textContent = picked.join(' ') || '…';
    box.innerHTML = '';
    var used = {};
    picked.forEach(function(w){ used[w] = (used[w] || 0) + 1; });
    var seen = {};
    pool.forEach(function(w, i){
      seen[w] = (seen[w] || 0) + 1;
      var isUsed = seen[w] <= (used[w] || 0);
      var b = el('button','wtok' + (isUsed ? ' used' : ''), w); b.type = 'button';
      b.addEventListener('click', function(){ picked.push(w); draw(); });
      box.appendChild(b);
    });
    if (picked.length === SENT[idx].length){
      var ok = picked.join(' ') === SENT[idx].join(' ');
      out.innerHTML = ok ? '<b style="color:var(--ok)">Correct.</b> ' + SENT[idx].join(' ') + '.'
                         : '<b style="color:var(--no)">Not yet.</b> Try again — look at which verb form fits the subject.';
      if (ok && !reduced()) line.style.borderColor = 'var(--ok)';
    } else {
      out.innerHTML = '<span style="color:var(--ink-2)">' + picked.length + ' of ' + SENT[idx].length + ' words placed.</span>';
      line.style.borderColor = 'var(--accent)';
    }
  }
  c.appendChild(line); c.appendChild(box);
  var row = el('div','labrow');
  var again = el('button','ghost','Start over'); again.type = 'button';
  again.addEventListener('click', function(){ picked = []; draw(); });
  var next = el('button','btn','Next sentence'); next.type = 'button';
  next.addEventListener('click', function(){ idx = (idx + 1) % SENT.length; load(); });
  row.appendChild(next); row.appendChild(again);
  c.appendChild(row); c.appendChild(out); load();
  return c;
}

function labTriangle(){
  var c = labCard('Pythagoras explorer', 'Change the two shorter sides and watch the hypotenuse, the squares and the area change.');
  var a = 6, b = 8, pic = el('div'), out = el('div','labout');
  function draw(){
    var h = Math.sqrt(a * a + b * b);

    pic.innerHTML = '<svg viewBox="0 0 300 170" class="dia" style="width:100%;max-width:320px">'
      + '<polygon points="40,150 ' + (40 + by).toFixed(1) + ',150 40,' + (150 - ax).toFixed(1) + '" class="dfill" fill-opacity="0.25"/>'
      + '<polyline points="40,150 ' + (40 + by).toFixed(1) + ',150 40,' + (150 - ax).toFixed(1) + ' 40,150" class="dacc"/>'
      + '<polyline points="56,150 56,134 40,134" class="dline"/>'
      + '<text x="' + (40 + by / 2).toFixed(1) + '" y="165" class="dmut" text-anchor="middle">' + b + '</text>'
      + '<text x="28" y="' + (150 - ax / 2).toFixed(1) + '" class="dmut" text-anchor="middle">' + a + '</text>'

      + '</svg>';
    out.innerHTML = 'h² = ' + a + '² + ' + b + '² = ' + (a * a) + ' + ' + (b * b) + ' = ' + (a * a + b * b)
      + '<br>h = √' + (a * a + b * b) + ' = <b>' + h.toFixed(3) + '</b>'
      + '<br>area of the triangle = ½ × ' + a + ' × ' + b + ' = <b>' + (a * b / 2) + '</b>'
      + (Math.abs(h - Math.round(h)) < 1e-9 ? '<br><span style="color:var(--ok)">' + a + ', ' + b + ', ' + h + ' is a Pythagorean triple.</span>' : '');
  }
  slider(c, 'Side a', 1, 20, a, 1, function(v){ a = v; draw(); });
  slider(c, 'Side b', 1, 20, b, 1, function(v){ b = v; draw(); });
  c.appendChild(pic); c.appendChild(out); draw();
  return c;
}

      body += '<line x1="' + sx(-6) + '" y1="' + sy(y).toFixed(1) + '" x2="' + sx(6) +
              '" y2="' + sy(y).toFixed(1) + '" class="gg"/>'
            + '<text x="' + (sx(0) - 6).toFixed(1) + '" y="' + (sy(y) + 3.5).toFixed(1) +
              '" class="gt" text-anchor="end">' + (y ? String(y).replace('-', '−') : '') + '</text>';
    body += '<line x1="' + sx(-6) + '" y1="' + sy(0) + '" x2="' + sx(6) + '" y2="' + sy(0) + '" class="ga"/>'
          + '<line x1="' + sx(0) + '" y1="' + sy(-8) + '" x2="' + sx(0) + '" y2="' + sy(12) + '" class="ga"/>'
          + '<path d="' + d + '" class="gc"/>';
    var txt;
    if (kind === 'line'){
      body += '<circle cx="' + sx(0) + '" cy="' + sy(k).toFixed(1) + '" r="4" class="gp"/>';
      var xint = m ? (-k / m) : null;
      if (xint !== null && xint >= -6 && xint <= 6)
        body += '<circle cx="' + sx(xint).toFixed(1) + '" cy="' + sy(0) + '" r="4" class="gp"/>';
      txt = 'y = ' + m + 'x ' + (k < 0 ? '− ' + Math.abs(k) : '+ ' + k) + '<br>'
          + 'gradient <b>' + m + '</b> (up ' + m + ' for every 1 across)<br>'
          + 'cuts the y-axis at <b>(0, ' + k + ')</b>'
          + (xint !== null ? ' and the x-axis at <b>(' + (Math.round(xint * 100) / 100) + ', 0)</b>' : '');
    } else {
      var disc = b * b - 4 * a * cc;
      var vx = -b / (2 * a), vy = a * vx * vx + b * vx + cc;
      body += '<circle cx="' + sx(vx).toFixed(1) + '" cy="' + sy(vy).toFixed(1) + '" r="4" class="gp"/>';
      txt = 'y = ' + a + 'x² ' + (b < 0 ? '− ' + Math.abs(b) : '+ ' + b) + 'x '
          + (cc < 0 ? '− ' + Math.abs(cc) : '+ ' + cc) + '<br>'
          + 'b² − 4ac = <b>' + disc + '</b> → ';
      if (disc > 0){
        var r1 = (-b - Math.sqrt(disc)) / (2 * a), r2 = (-b + Math.sqrt(disc)) / (2 * a);
        txt += 'two roots: <b>x = ' + (Math.round(r1 * 100) / 100) + '</b> and <b>x = ' + (Math.round(r2 * 100) / 100) + '</b>';
        [r1, r2].forEach(function(r){
          if (r >= -6 && r <= 6) body += '<circle cx="' + sx(r).toFixed(1) + '" cy="' + sy(0) + '" r="4" class="gp"/>';
        });
      } else if (disc === 0){ txt += 'one repeated root at <b>x = ' + (Math.round(vx * 100) / 100) + '</b>'; }
      else { txt += '<b>no real roots</b> — the curve misses the x-axis'; }
      txt += '<br>turning point <b>(' + (Math.round(vx * 100) / 100) + ', ' + (Math.round(vy * 100) / 100) + ')</b>, '
          + (a > 0 ? 'a minimum (curve opens upwards)' : 'a maximum (curve opens downwards)');
    }
    pic.innerHTML = '<svg viewBox="0 0 300 190" class="graph" style="width:100%;max-width:340px">' + body + '</svg>';
    out.innerHTML = txt;
  }
  function controls(){
    rows.innerHTML = '';
    if (kind === 'line'){
      slider(rows, 'gradient m', -5, 5, m, 1, function(v){ m = v; draw(); });
      slider(rows, 'intercept c', -6, 6, k, 1, function(v){ k = v; draw(); });
    } else {
      slider(rows, 'a', -3, 3, a, 1, function(v){ a = v || 1; draw(); });
      slider(rows, 'b', -6, 6, b, 1, function(v){ b = v; draw(); });
      slider(rows, 'c', -8, 8, cc, 1, function(v){ cc = v; draw(); });
    }
  }
  var seg = el('div','seg');
  [['line','Straight line'], ['quad','Quadratic']].forEach(function(p){
    var btn = el('button', null, p[1]); btn.type = 'button';
    btn.setAttribute('aria-pressed', kind === p[0] ? 'true' : 'false');
    btn.addEventListener('click', function(){
      kind = p[0];
      var kids = seg.children;
      for (var i = 0; i < kids.length; i++) kids[i].setAttribute('aria-pressed','false');
      btn.setAttribute('aria-pressed','true');
      controls(); draw();
    });
    seg.appendChild(btn);
  });
  var top = el('div','labrow'); top.appendChild(seg);
  c.appendChild(top); c.appendChild(rows); c.appendChild(pic); c.appendChild(out);
  controls(); draw();
  return c;
}
